async function generateContent() { const trend = document.getElementById('trendInput').value; const errorDiv = document.getElementById('error'); errorDiv.innerHTML = ''; try { const response = await fetch('/.netlify/functions/main-orchestrator', { method: 'POST', body: JSON.stringify({ trend: trend }) }); const data = await response.json(); if (data.success) { // GUARDAR en el navegador let history = JSON.parse(localStorage.getItem('shibuyaHistory') || '[]'); history.push({ fecha: new Date().toLocaleDateString('es-PE'), hora: new Date().toLocaleTimeString('es-PE'), tendencia: trend, contenido: data.content, score: data.score }); localStorage.setItem('shibuyaHistory', JSON.stringify(history)); // MOSTRAR errorDiv.innerHTML = '
✅ Contenido generado y guardado
'; document.getElementById('lastRun').textContent = new Date().toLocaleString('es-PE'); document.getElementById('contentCount').textContent = history.length; // MOSTRAR HISTORIAL updateHistory(); } } catch (error) { errorDiv.innerHTML = '
Error: ' + error.message + '
'; } } function updateHistory() { let history = JSON.parse(localStorage.getItem('shibuyaHistory') || '[]'); const historyDiv = document.getElementById('history'); if (history.length === 0) { historyDiv.innerHTML = '

Esperando primer contenido...

'; return; } historyDiv.innerHTML = history.map(item => `
${item.fecha} ${item.hora}
Tendencia: ${item.tendencia}
Contenido: ${item.contenido}
Score: ${item.score}
`).reverse().join(''); } // MOSTRAR HISTORIAL AL CARGAR document.addEventListener('DOMContentLoaded', updateHistory);
